cmake_minimum_required(VERSION 3.21)
project(Kassa VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets)
if(MSVC)
    add_compile_options(/utf-8 /W4)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(kassa_widgets STATIC
    kassa/mainwindow.cpp kassa/mainwindow.h kassa/mainwindow.ui
    kassa/inputdialog.cpp kassa/inputdialog.h kassa/inputdialog.ui
    kassa/delegate.cpp kassa/delegate.h kassa/delegate.ui kassa/money.h)
target_include_directories(kassa_widgets PUBLIC kassa)
target_link_libraries(kassa_widgets PUBLIC Qt${QT_VERSION_MAJOR}::Widgets)
add_executable(Kassa WIN32 kassa/main.cpp)
target_link_libraries(Kassa PRIVATE kassa_widgets)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Test)
    add_executable(kassa_tests tests/tst_kassa.cpp)
    target_link_libraries(kassa_tests PRIVATE kassa_widgets Qt${QT_VERSION_MAJOR}::Test)
    add_test(NAME kassa_tests COMMAND kassa_tests)
    set_tests_properties(kassa_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME launch COMMAND Kassa --smoke-test)
    set_tests_properties(launch PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 15)
endif()
